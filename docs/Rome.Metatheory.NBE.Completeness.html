-- Consistency implies completeness.

completeness : ∀ {τ υ : Type Δ κ} →
                ⇓ₜ τ ≡ ⇓ₜ υ → τ ≡ₜ υ 
completeness {τ = τ} {υ} e = 
   τ        ≡ₜ⟨ consistency τ ⟩ 
   ⇑ (⇓ₜ τ)  ≡ₜ⟨ inst (cong ⇑ e) ⟩ 
   ⇑ (⇓ₜ υ)  ≡ₜ⟨ consistency υ ⁻¹ ⟩ 
   υ ∎
