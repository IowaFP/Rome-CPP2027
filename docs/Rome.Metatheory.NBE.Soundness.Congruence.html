--------------------------------------------------------------------------------
-- - Uniformity is preserved under renaming (ren-Uniform)
--   (This is actually just what uniformity means.)

ren-Uniform : ∀ {κ₁ : Kind} {κ₂ : Kind} {F : KripkeFunction Δ₁ κ₁ κ₂} → (ρ : Renaming Δ₁ Δ₂) → Uniform F → Uniform (renKripke ρ F) 
ren-Uniform ρ Unif-F ρ₁ ρ₂ V₁ V₂ q = Unif-F (ρ₁ ∘ ρ) ρ₂ V₁ V₂ q

--------------------------------------------------------------------------------
-- Application respects ≈

cong-App : ∀ {Δ : KEnv} {κ₁ : Kind} {κ₂ : Kind} {V₁ V₂ : SemType Δ (κ₁ `→ κ₂)} → 
           _≈_ {κ = κ₁ `→ κ₂} V₁ V₂ → 
           {W₁ W₂ : SemType Δ κ₁} → 
           W₁ ≈ W₂ → 
           (V₁ ⟨·⟩ W₁) ≈ (V₂ ⟨·⟩ W₂)
cong-App {V₁ = F} {G} (unif-F , unif-G , Ext) q = Ext id q           

--------------------------------------------------------------------------------
-- Mapping respects ≈

cong-$ : ∀ {V₁ V₂ : SemType Δ (κ₁ `→ κ₂)} → 
           _≈_ {κ = κ₁ `→ κ₂} V₁ V₂ → 
           {W₁ W₂ : SemType Δ R[ κ₁ ]} → 
           _≈_ {κ = R[ κ₁ ]} W₁ W₂ → 
           _≈_ {κ = R[ κ₂ ]} (V₁ ⟨$⟩ W₁)  (V₂ ⟨$⟩ W₂)
cong-$ v {l₁ ▹ τ₁} {l₂ ▹ τ₂} ((refl , refl) , rel) = (refl , refl) , (cong-App v rel)
cong-$ v {row (n , P) _} {row (m , Q) _} (refl , eq) =  refl , λ { i → eq i .fst , cong-App v (eq i .snd) }
cong-$ v {ρ₂ ∖ ρ₁} {ρ₄ ∖ ρ₃} (rel₁ , rel₂) = (cong-$ v rel₁) , (cong-$ v rel₂)
cong-$ {V₁ = F} {G} (Unif-F , Unif-G , Ext-FG) {φ₁ $ n₁} {φ₂ $ n₂} ((refl , Unif-φ₁ , Unif-φ₂ , Ext-φ) , (refl , refl)) = 
  (refl , 
  (λ r₁ r₂ V → trans-≈ 
    (Unif-F r₁ r₂ (φ₁ r₁ V) (φ₁ r₁ V) (trans-≈ (Ext-φ r₁ V) (sym-≈ (Ext-φ r₁ V)))) 
    (refl-Extₗ Ext-FG (r₂ ∘ r₁) (Unif-φ₁ r₁ r₂ V))) , 
  (λ r₁ r₂ V → trans-≈ 
    (Unif-G r₁ r₂ (φ₂ r₁ V) (φ₂ r₁ V) (trans-≈ (sym-≈ (Ext-φ r₁ V)) (Ext-φ r₁ V))) 
    (refl-Extᵣ Ext-FG (r₂ ∘ r₁) (Unif-φ₂ r₁ r₂ V))) , 
  (λ r V → Ext-FG r (Ext-φ r V))) , 
  refl , refl

--------------------------------------------------------------------------------
-- Given a : κ₁, The semantic image of (λ f : κ₁ `→ κ₂. f a) is uniform.
-- (This goal appears with the use of the flapping operator (??).)

Unif-apply : ∀ {Δ : KEnv} {κ₁ : Kind} {κ₂ : Kind} {V₁ V₂ : SemType Δ κ₁} → 
               V₁ ≈ V₂ → 
               Uniform {Δ = Δ} {κ₁ `→ κ₂} {κ₂} (apply V₂)
Unif-apply {V₁ = V₁} {V₂} v ρ₁ ρ₂ V₃ V₄ x = 
  trans-≈
    (fst x id ρ₂ (renSem ρ₁ V₂) (renSem ρ₁ V₂) (ren-≈ ρ₁ (refl-≈ʳ v)))
    (third x ρ₂ (sym-≈ (renSem-comp-≈ ρ₁ ρ₂ (refl-≈ʳ v)))) 

cong-apply : ∀ {Δ : KEnv} {κ₁ : Kind} {κ₂ : Kind} {V₁ V₂ : SemType Δ κ₁} → 
               V₁ ≈ V₂ → 
               _≈_ {κ = (κ₁ `→ κ₂) `→ κ₂} (apply V₁)  (apply V₂)
cong-apply v = 
  Unif-apply (sym-≈ v) , 
  Unif-apply v , 
  λ ρ v' → third v' id (ren-≈ ρ v)  

--------------------------------------------------------------------------------
-- Flapping respects ≈

cong-<?> : ∀ {Δ : KEnv} {κ₁ : Kind} {κ₂ : Kind} {V₁ V₂ : SemType Δ R[ κ₁ `→ κ₂ ]} → 
           _≈_ {κ = R[ κ₁ `→ κ₂ ]} V₁ V₂ → 
           {W₁ W₂ : SemType Δ κ₁} → 
           _≈_ {κ = κ₁} W₁ W₂ → 
           _≈_ {κ = R[ κ₂ ]} (V₁ ⟨?⟩ W₁)  (V₂ ⟨?⟩ W₂)
cong-<?> v {W₁} {W₂} w = 
  cong-$ 
  (cong-apply w) v

-- -- Congruence over complements

cong-compl : {Δ : KEnv} {κ : Kind} {n m : ℕ} 
             (A B : Fin n → Label × SemType Δ κ)
             (C D : Fin m → Label × SemType Δ κ) → 
             ((i : Fin n) → A i ≈₂ B i) → 
             ((i : Fin m) → C i ≈₂ D i) → 
             compl A C ≈ᵣ compl B D
cong-compl {n = zero} A B C D i₁ i₂ = refl , λ ()
cong-compl {n = suc n} A B C D i₁ i₂ with i₁ fzero | A fzero .fst ∈Row? C | B fzero .fst ∈Row? D 
... | eq , rel  | yes p         | yes q = cong-compl (A ∘ fsuc) (B ∘ fsuc) C D (i₁ ∘ fsuc) i₂
... | eq₁ , rel | yes (j , eq₂) | no q = ⊥-elim (q (j , (trans (trans (sym eq₁) eq₂) (i₂ j .fst))))
... | eq₁ , rel | no  q         | yes (j , eq₂) = ⊥-elim (q (j , (trans (trans eq₁ eq₂) (sym (i₂ j .fst)))))
... | eq , rel  | no  p         | no q  with 
      (compl (A ∘ fsuc) C)
    | (compl (B ∘ fsuc) D)
    | cong-compl (A ∘ fsuc) (B ∘ fsuc) C D (i₁ ∘ fsuc) i₂
... | n₁ , P | n₂ , Q | (refl , P≈Q) = 
  refl , (λ { fzero    → i₁ fzero ; 
              (fsuc i) → P≈Q i })

cong-∖v : ∀ {Δ : KEnv} {κ : Kind} {V₁ V₂ W₁ W₂ : Row (SemType Δ κ)} → 
           V₂ ≈ᵣ W₂ → 
           V₁ ≈ᵣ W₁ → 
           (V₂ ∖v V₁) ≈ᵣ (W₂ ∖v W₁)
cong-∖v {V₁ = n , P} {m , Q} {l , R} {j , I} (refl , v₂) (refl , v₁) = cong-compl Q I P R v₂ v₁ 

cong-⟨∖⟩ : ∀ {V₁ V₂ W₁ W₂ : SemType Δ R[ κ ]} → 
           V₂ ≈ W₂ → 
           V₁ ≈ W₁ → 
           (V₂ ⟨∖⟩ V₁) ≈ (W₂ ⟨∖⟩ W₁)

cong-⟨∖⟩ {V₁ = x₁ ▹ x₂} {x₃ ▹ x₄} {x₅ ▹ x₆} {x₇ ▹ x₈} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = x₁ ▹ x₂} {row ρ x₃} {x₄ ▹ x₅} {row ρ₁ x₆} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = x₁ ▹ x₂} {V₂ ∖ V₃} {x₃ ▹ x₄} {W₂ ∖ W₃} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = x₁ ▹ x₂} {V₂ $ V₃} {x₃ ▹ x₄} {W₂ $ W₃} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = row ρ x₁} {x₂ ▹ x₃} {row ρ₁ x₄} {x₅ ▹ x₆} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = row ρ x₁} {row ρ₁ x₂} {row ρ₂ x₃} {row ρ₃ x₄} rel₁ rel₂ = cong-∖v rel₁ rel₂
cong-⟨∖⟩ {V₁ = row ρ x₁} {V₂ ∖ V₃} {row ρ₁ x₂} {W₂ ∖ W₃} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = row ρ x₁} {V₂ $ V₃} {row ρ₁ x₂} {W₂ $ W₃} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = V₁ ∖ V₂} {x₁ ▹ x₂} {W₁ ∖ W₂} {x₃ ▹ x₄} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = V₁ ∖ V₂} {row ρ x₁} {W₁ ∖ W₂} {row ρ₁ x₂} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = V₁ ∖ V₂} {V₃ ∖ V₄} {W₁ ∖ W₂} {W₃ ∖ W₄} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = V₁ ∖ V₂} {_ $ _} {W₁ ∖ W₂} {_ $ _} rel₁ rel₂ = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = φ₁ $ n₁} {ψ₁ $ x₁} {φ₂ $ n₂} {ψ₂ $ x₂} rel₁ rel₂@((refl , Unif-φ₁ , Unif-φ₂ , Ext) , (refl , refl)) = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = φ₁ $ n₁} {x₁ ▹ x₂} {φ₂ $ n₂} {x₃ ▹ x₄} rel₁ rel₂@((refl , Unif-φ₁ , Unif-φ₂ , Ext) , (refl , refl)) = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = φ₁ $ n₁} {row ρ x₁} {φ₂ $ n₂} {row ρ₁ x₂} rel₁ rel₂@((refl , Unif-φ₁ , Unif-φ₂ , Ext) , (refl , refl)) = rel₁ , rel₂
cong-⟨∖⟩ {V₁ = φ₁ $ n₁} {x₁ ∖ x₂} {φ₂ $ n₂} {y₁ ∖ y₂} rel₁ rel₂@((refl , Unif-φ₁ , Unif-φ₂ , Ext) , (refl , refl)) = rel₁ , rel₂
