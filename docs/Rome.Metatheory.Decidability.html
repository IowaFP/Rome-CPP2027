--------------------------------------------------------------------------------
-- -- Type equivalence is decidable

_≡ₜ?_ : ∀ (τ₁ τ₂ : Type Δ κ) → Dec (τ₁ ≡ₜ τ₂)
τ₁ ≡ₜ? τ₂  with (⇓ₜ τ₁) ≡? (⇓ₜ τ₂)
... | yes p = yes (completeness p)
... | no  p = no (p ∘ soundness)
