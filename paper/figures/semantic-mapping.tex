_.map.'_ : SemType Delta (kappa_1 '-> kappa_2) -> SemType Delta R[ kappa_1 ] -> 
       SemType Delta R[ kappa_2 ]
phi .map.' (n := tau) = n := (phi cdot' tau)
phi .map.' (row (n , P) q) = row (n , fmap (phi cdot'_) . P)
phi .map.' (rho_2 - rho_1) = (phi .map. rho_2) - (phi .map. rho_1)
phi_1 .map.' (phi_2 .map. n) = (\ r -> phi_1 r . phi_2 r) .map. n
